# 智能排版 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 按已定稿 design 交付 Yooco「智能排版」P0：侧栏进入 `/layout.html`，三区网格工作台，3 模板 × 3 比例，DeepSeek 填槽生成可读预览，生成后可拖块并按可调列数 N 吸附，支持换槽重排覆盖拖动；只预览不导出。

**Architecture:** 独立静态页 `public/layout.html`（同工作室壳，不进 `/studio.html`「换一版」）。前端用百分比槽坐标渲染画布；纯函数模块负责网格行数推导与吸附。新 API `POST /api/layout-fill` 走现有 DeepSeek 通道与 `prepareCreditSpend` / `spendCredits`（成功才扣，动作价与工作台共用 `layout` 额度）。合进 `master` 后 `public/*` 经 EdgeOne `npm run build:edgeone`（`next build`）自动上线。

**Tech Stack:** 静态 HTML/CSS/JS（`public/`）、Next App Router API（`app/api/layout-fill`）、DeepSeek Chat Completions（`DEEPSEEK_API_KEY` / `getDeepseekModel`）、现有积分桶（`lib/credits.js` + `lib/edgeone-rate-limit.ts`）、Node 可跑校验脚本（`scripts/check-layout-*.mjs`）。

## Global Constraints

- 合进 master = EdgeOne 自动上线；合前要截图给阿超；阿超说「合并」才能合
- 云端代码 grok-4.7、fast=false；commit 中文
- P0：只预览不导出；3 模板；比例 3:4/1:1/16:9；拖块+网格吸附；格子 N 默认 12 可调 6–24；换槽重排覆盖拖动；不复用公众号 normalize 提示词；与 PR #70 分离

- 静态页与脚本放 `public/`：`layout.html`、`layout.css`、`layout.js`、`layout-templates.js`、`layout-grid.js`

```js
export function clampGridN(n) {
  const value = Math.round(Number(n));
  if (!Number.isFinite(value)) return 12;
  return Math.min(24, Math.max(6, value));
}

export function gridRows(n, ratioW, ratioH) {
  const cols = clampGridN(n);
  const w = Number(ratioW) || 1;
  const h = Number(ratioH) || 1;
  return Math.max(1, Math.round((cols * h) / w));
}

export function snapToGrid(xPct, yPct, n, m) {
  const cols = Math.max(1, n);
  const rows = Math.max(1, m);
  const stepX = 100 / cols;
  const stepY = 100 / rows;
  const gx = Math.round(xPct / stepX) * stepX;
  const gy = Math.round(yPct / stepY) * stepY;
  // 半格阈值：round 已等价于「距交点 ≤ 半格则吸」
  return {
    x: Math.min(100, Math.max(0, gx)),
    y: Math.min(100, Math.max(0, gy)),
  };
}

export function clampBlock(x, y, w, h) {
  const width = Math.max(1, w);
  const height = Math.max(1, h);
  const minX = -width + 1;
  const minY = -height + 1;
  const maxX = 99;
  const maxY = 99;
  return {
    x: Math.min(maxX, Math.max(minX, x)),
    y: Math.min(maxY, Math.max(minY, y)),
  };
}

    id,
    pointerId: event.pointerId,
    offsetX: ((event.clientX - rect.left) / rect.width) * 100 - box.x,
    offsetY: ((event.clientY - rect.top) / rect.height) * 100 - box.y,
    w: box.w,
    h: box.h,

  block.setPointerCapture(event.pointerId);
});

  if (!drag || event.pointerId !== drag.pointerId) return;
  const rect = canvas.getBoundingClientRect();
  let x = ((event.clientX - rect.left) / rect.width) * 100 - drag.offsetX;
  let y = ((event.clientY - rect.top) / rect.height) * 100 - drag.offsetY;

  const m = gridRows(n, ratio.w, ratio.h);

  const clamped = clampBlock(snapped.x, snapped.y, drag.w, drag.h);
  state.positions[drag.id] = { x: clamped.x, y: clamped.y };
  const el = canvasBlocks.querySelector(`[data-id="${drag.id}"]`);
  if (el) {
    el.style.left = `${clamped.x}%`;
    el.style.top = `${clamped.y}%`;
  }
});

function endDrag(event) {
  if (!drag || event.pointerId !== drag.pointerId) return;
  const el = canvasBlocks.querySelector(`[data-id="${drag.id}"]`);

import { AppError } from "./deepseek-normalizer.js";

const MAX_SOURCE_CHARS = 20_000;
const FILL_TIMEOUT_MS = 60_000;
const RATIO_IDS = ["3:4", "1:1", "16:9"];
const TEMPLATE_IDS = ["t1", "t2", "t3"];
const SLOT_TYPES = ["H1", "H2", "body", "image"];

function charLength(value) {
  return Array.from(String(value || "")).length;
}

function splitParagraphs(source) {
  return String(source || "")
    .replace(/\r\n/g, "\n")
    .split(/\n+/)
    .map((part) => part.trim())
    .filter(Boolean);
}

export function validateLayoutFillRequest(body) {
  if (!body || typeof body.source !== "string") {
    throw new AppError("INVALID_INPUT", "请先贴文案。", 400);
  }
  const source = body.source.replace(/\r\n/g, "\n").trim();
  if (!source) throw new AppError("EMPTY_INPUT", "请先贴文案。", 400);
  if (charLength(source) > MAX_SOURCE_CHARS) {
    throw new AppError("SOURCE_TOO_LARGE", "文案过长，请控制在 20000 字以内。", 413);
  }
  const imageCount = Number(body.imageCount);
  if (!Number.isInteger(imageCount) || imageCount < 0 || imageCount > 20) {
    throw new AppError("INVALID_INPUT", "图片张数不合法。", 400);
  }
  const ratio = String(body.ratio || "");
  if (!RATIO_IDS.includes(ratio)) throw new AppError("INVALID_INPUT", "比例只能是 3:4、1:1 或 16:9。", 400);
  const templateId = String(body.templateId || "");
  if (!TEMPLATE_IDS.includes(templateId)) throw new AppError("INVALID_INPUT", "模板不存在。", 400);
  if (!Array.isArray(body.slots) || body.slots.length < 1) {
    throw new AppError("INVALID_INPUT", "缺少槽清单。", 400);
  }
  const slots = body.slots.map((item) => {
    const id = String(item?.id || "");
    const type = String(item?.type || "");
    if (!id || !SLOT_TYPES.includes(type)) {
      throw new AppError("INVALID_INPUT", "槽定义不合法。", 400);
    }
    return { id, type };
  });
  const seed = body.seed == null ? undefined : Number(body.seed);
  if (seed != null && !Number.isInteger(seed)) {
    throw new AppError("INVALID_INPUT", "seed 必须是整数。", 400);
  }
  return { source, imageCount, ratio, templateId, slots, seed };
}

export function defaultFill(ctx) {
  const parts = splitParagraphs(ctx.source);
  const textSlots = ctx.slots.filter((slot) => slot.type !== "image");
  const imageSlots = ctx.slots.filter((slot) => slot.type === "image");
  const filled = [];
  textSlots.forEach((slot, index) => {
    if (slot.type === "H1") filled.push({ id: slot.id, type: slot.type, text: parts[0] || "" });
    else if (slot.type === "H2") filled.push({ id: slot.id, type: slot.type, text: parts[1] || "" });
    else {
      const bodyParts = parts.slice(textSlots[0]?.type === "H1" ? (textSlots[1]?.type === "H2" ? 2 : 1) : 0);
      // 多 body：按 body 序号取剩余段
      const bodyIndex = textSlots.filter((item) => item.type === "body").indexOf(slot);
      filled.push({ id: slot.id, type: slot.type, text: bodyParts[bodyIndex] || bodyParts.join("\n") || "" });
    }
  });
  imageSlots.forEach((slot, index) => {
    if (index < ctx.imageCount) filled.push({ id: slot.id, type: "image", imageIndex: index });
    else filled.push({ id: slot.id, type: "image" }); // 空着，无 imageIndex
  });
  return {
    templateId: ctx.templateId,
    ratio: ctx.ratio,
    seed: ctx.seed ?? null,
    slots: filled,
  };
}

function validImageIndex(value, imageCount) {
  return Number.isInteger(value) && value >= 0 && value < imageCount;
}

export function normalizeFillResult(raw, ctx) {
  const byId = new Map(ctx.slots.map((slot) => [slot.id, slot]));
  if (!raw || !Array.isArray(raw.slots)) return defaultFill(ctx);
  const out = [];
  let illegal = false;
  for (const slot of ctx.slots) {
    const hit = raw.slots.find((item) => item && item.id === slot.id);
    if (!hit || hit.type !== slot.type) {
      illegal = true;
      break;
    }
    if (slot.type === "image") {
      if (ctx.imageCount === 0) {
        out.push({ id: slot.id, type: "image" });
      } else if (hit.imageIndex == null) {
        out.push({ id: slot.id, type: "image" });
      } else if (!validImageIndex(hit.imageIndex, ctx.imageCount)) {
        illegal = true;
        break;
      } else {
        out.push({ id: slot.id, type: "image", imageIndex: hit.imageIndex });
      }
    } else {
      out.push({ id: slot.id, type: slot.type, text: String(hit.text || "") });
    }
  }
  for (const item of raw.slots || []) {
    if (item?.id && !byId.has(item.id)) illegal = true;
  }
  if (illegal) return defaultFill(ctx);
  return {
    templateId: ctx.templateId,
    ratio: ctx.ratio,
    seed: Number.isInteger(raw.seed) ? raw.seed : (ctx.seed ?? null),
    slots: out,
  };
}

  }
  const fetchImpl = options.fetchImpl || fetch;
  const controller = new AbortController();
  const timeout = setTimeout(() => controller.abort(), options.timeoutMs || FILL_TIMEOUT_MS);
  let response;
  try {
    response = await fetchImpl("https://api.deepseek.com/chat/completions", {
      method: "POST",
      signal: controller.signal,
      headers: {
        "content-type": "application/json",
        authorization: `Bearer ${apiKey}`,
      },
      body: JSON.stringify({
        model,
        temperature: ctx.seed == null ? 0.2 : 0.7,
        stream: false,
        response_format: { type: "json_object" },
        messages: [
          { role: "system", content: buildFillPrompt(ctx) },
          { role: "user", content: ctx.source },
        ],
      }),
    });
  } catch (error) {
    if (error?.name === "AbortError") {
      throw new AppError("DEEPSEEK_TIMEOUT", "排版失败，再试一次", 504);
    }
    throw new AppError("DEEPSEEK_UNREACHABLE", "排版失败，再试一次", 502);
  } finally {
    clearTimeout(timeout);
  }

  if (!response.ok) {
    throw new AppError("DEEPSEEK_REQUEST_FAILED", "排版失败，再试一次", response.status === 429 ? 429 : 502);
  }

  let raw;
  try {
    const payload = await response.json();
    raw = JSON.parse(payload?.choices?.[0]?.message?.content || "");
  } catch {
    throw new AppError("DEEPSEEK_INVALID_OUTPUT", "排版失败，再试一次", 502);
  }
  return normalizeFillResult(raw, ctx);
}

export { AppError };
```

注意：超时/坏 JSON 的用户文案必须是 **「排版失败，再试一次」**（design §6/§7）。校验失败走 `defaultFill`，仍算成功响应（可扣次）——与「坏 JSON」区分：坏 JSON 在 parse 阶段抛错不扣次。

```ts
import { creditCost } from "../../../lib/credits";
import {
  AppError,
  fillLayoutSlots,
  validateLayoutFillRequest,
} from "../../../lib/layout-fill";
import {
  prepareCreditSpend,
  RateLimitError,
  spendCredits,
  TRIAL_LIMIT,
  UPGRADE_OFFER,
  type CreditSnapshot,
} from "../../../lib/edgeone-rate-limit";
import { getDeepseekApiKey, getDeepseekModel } from "../../../lib/runtime-env";

const MAX_BODY_BYTES = 512 * 1024;
export const dynamic = "force-dynamic";

function headers(): Record<string, string> {
  return { "cache-control": "no-store" };
}

export async function POST(request: Request) {
  try {
    const contentLength = Number(request.headers.get("content-length") || 0);
    if (contentLength > MAX_BODY_BYTES) {
      throw new AppError("REQUEST_TOO_LARGE", "提交内容过大。", 413);
    }
    const body = await request.json();
    const ctx = validateLayoutFillRequest(body);
    const cost = creditCost("layout"); // 共用工作台额度
    if (!cost) throw new AppError("INTERNAL_ERROR", "排版积分价格没配好。", 500);
    const creditUser = await prepareCreditSpend(request, cost);
    const data = await fillLayoutSlots(ctx, getDeepseekApiKey(), getDeepseekModel());
    const trial = await spendCredits(creditUser, "layout", cost);
    return Response.json({ ok: true, data, trial, credits: trial.credits }, { headers: headers() });
  } catch (error) {
    const known = error instanceof AppError;
    const status = known ? error.status : 500;
    const code = known ? error.code : "INTERNAL_ERROR";
    const message = known ? error.message : "排版失败，再试一次";
    const responseHeaders = headers();
    const payload: {
      ok: false;
      credits?: CreditSnapshot;
      error: {
        code: string;
        message: string;
        remaining?: number;
        limit?: number;
        bonus?: number;
        credits?: CreditSnapshot;
        upgrade?: typeof UPGRADE_OFFER;
      };
    } = { ok: false, error: { code, message } };
    if (error instanceof RateLimitError) {
      responseHeaders["retry-after"] = String(error.retryAfter);
      payload.error.remaining = error.remaining;
      payload.error.limit = error.limit;
      payload.error.bonus = error.bonus;
      payload.credits = {
        remaining: error.remaining,
        limit: error.limit,
        bonus: error.bonus,
        unlimited: false,
      };
      payload.error.credits = payload.credits;
      if (error.code === "TRIAL_EXHAUSTED") payload.error.upgrade = UPGRADE_OFFER;
    } else {
      payload.error.limit = TRIAL_LIMIT;
    }
    return Response.json(payload, { status, headers: responseHeaders });
  }
}
```

在 `lib/credits.js` 顶部注释补一行：`layout` 同时用于工作室优化与智能排版填槽（共用日额度）。
